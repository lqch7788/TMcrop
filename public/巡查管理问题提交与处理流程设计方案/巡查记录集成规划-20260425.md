**3.2 实现完整渲染逻辑**

```typescript
export function InspectionTab(props: InspectionTabProps) {
  const {
    filters, onFilterChange, onResetFilters,
    currentPage, pageSize, onPageChange, onPageSizeChange,
    exportMode, batchEditMode, batchDeleteMode,
    onToggleExportMode, onToggleBatchEditMode, onToggleBatchDeleteMode,
    selectedRows, onToggleSelectRow, onSelectAll, onClearSelection,
    filteredRecords, allRecords,
    detailRecordId, onViewDetail, onCloseDetail,
    isCreateModalOpen, onOpenCreateModal, onCloseCreateModal,
    problems, onReportProblem, onAcceptProblem,
    onBatchDelete, onBatchEdit,
  } = props;

  // 详情记录
  const detailRecord = useMemo(() =>
    allRecords.find(r => r.id === detailRecordId) || null,
    [allRecords, detailRecordId]
  );

  // 分页后的数据
  const paginatedRecords = useMemo(() => {
    const start = (currentPage - 1) * pageSize;
    return filteredRecords.slice(start, start + pageSize);
  }, [filteredRecords, currentPage, pageSize]);

  // 选中ID转字符串数组
  const selectedIds = useMemo(() =>
    selectedRows.map(idx => paginatedRecords[idx]?.id).filter(Boolean),
    [selectedRows, paginatedRecords]
  );

  return (
    <div className="space-y-4">
      {/* 搜索栏 */}
      <InspectionSearch
        filters={filters}
        onFiltersChange={onFilterChange}
        onReset={onResetFilters}
      />

      {/* 工具栏 */}
      <InspectionToolbar
        exportMode={exportMode}
        batchEditMode={batchEditMode}
        batchDeleteMode={batchDeleteMode}
        selectedCount={selectedRows.length}
        onCreate={onOpenCreateModal}
        onExport={onToggleExportMode}
        onBatchEdit={onToggleBatchEditMode}
        onBatchDelete={onToggleBatchDeleteMode}
        onConfirmExport={() => {/* 导出逻辑 */}}
        onConfirmBatchEdit={() => {/* 批量编辑 */}}
        onConfirmBatchDelete={() => onBatchDelete(selectedIds)}
        onCancel={() => {
          onToggleExportMode();
          onToggleBatchEditMode();
          onToggleBatchDeleteMode();
          onClearSelection();
        }}
      />

      {/* 表格 */}
      <InspectionTable
        records={paginatedRecords}
        selectedRows={selectedRows}
        onSelectRow={onToggleSelectRow}
        onSelectAll={onSelectAll}
        onViewDetail={onViewDetail}
        problems={problems}
        onAccept={onAcceptProblem}
      />

      {/* 分页 */}
      <div className="flex justify-between items-center px-4 py-3 border-t">
        <span className="text-sm text-gray-500">
          共 {filteredRecords.length} 条记录
        </span>
        <div className="flex gap-2">
          <select
            value={pageSize}
            onChange={(e) => onPageSizeChange(Number(e.target.value))}
            className="px-2 py-1 border rounded"
          >
            <option value={10}>10条/页</option>
            <option value={20}>20条/页</option>
            <option value={50}>50条/页</option>
          </select>
          <button
            onClick={() => onPageChange(currentPage - 1)}
            disabled={currentPage === 1}
            className="px-3 py-1 border rounded disabled:opacity-50"
          >
            上一页
          </button>
          <span className="px-3 py-1">
            第 {currentPage}/{Math.ceil(filteredRecords.length / pageSize)} 页
          </span>
          <button
            onClick={() => onPageChange(currentPage + 1)}
            disabled={currentPage >= Math.ceil(filteredRecords.length / pageSize)}
            className="px-3 py-1 border rounded disabled:opacity-50"
          >
            下一页
          </button>
        </div>
      </div>

      {/* 弹窗 */}
      {isCreateModalOpen && (
        <CreateInspectionModal
          isOpen={isCreateModalOpen}
          onClose={onCloseCreateModal}
          onSubmit={() => {
            onCloseCreateModal();
            // 刷新数据
          }}
          // ... 其他props
        />
      )}

      {detailRecord && (
        <DetailInspectionModal
          isOpen={!!detailRecord}
          onClose={onCloseDetail}
          record={detailRecord}
          onReportProblem={() => onReportProblem(detailRecord)}
        />
      )}
    </div>
  );
}
```
