# 作物管理模块实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 在V1.1系统中实现作物管理模块，包含种源管理、育苗管理、种植管理三个子模块，保留旧系统全部功能，UI风格与农事管理模块保持一致。

**架构：** 新增"作物管理"主菜单，下设种源管理、育苗管理、种植管理三个页面。采用组件化开发模式，所有数据从types和data导入，使用UnifiedModal弹窗，emerald绿色主题UI。

**技术栈：** React + TypeScript + Ant Design + Tailwind CSS + qrcode.react

```typescript
        cropVarieties={cropVarieties}

          onPrint={handlePrint}

      )}

        isOpen={lightboxOpen}
```

```typescript
// src/components/farm/seed-source/modals/PrintLabelModal.tsx

import React from 'react';
import { QRCodeSVG } from 'qrcode.react';
import { UnifiedModal } from '../../../ui/UnifiedModal';
import { SeedSource } from '../../../../types/crop';

interface PrintLabelModalProps {
  isOpen: boolean;
  onClose: () => void;
  record: SeedSource;
}

export function PrintLabelModal({ isOpen, onClose, record }: PrintLabelModalProps) {
  const handlePrint = () => {
    window.print();
  };

  const qrCodeValue = JSON.stringify({
    type: 'seed-source',
    code: record.seedCode,
    date: record.purchaseDate
  });

  return (
    <UnifiedModal
      isOpen={isOpen}
      onClose={onClose}
      title="打印标签"
      size="md"
      showFooter={true}
      onFooterConfirm={handlePrint}
      onFooterCancel={onClose}
    >
      <div className="print:hidden">
        {/* 标签预览 */}
        <div className="border-2 border-dashed border-gray-300 rounded-lg p-6 mb-4">
          <div className="flex flex-col items-center">
            {/* 二维码 */}
            <div className="bg-white p-2 border border-gray-200 rounded">
              <QRCodeSVG value={qrCodeValue} size={120} />
            </div>

            {/* 标签信息 */}
            <div className="mt-4 text-center">
              <div className="text-lg font-bold text-gray-900">{record.seedCode}</div>
              <div className="text-sm text-gray-600">{record.cropName}</div>
              <div className="text-sm text-gray-600">
                {record.availableCount.toLocaleString()} {record.unit}
              </div>
              <div className="text-xs text-gray-400 mt-2">{record.purchaseDate}</div>
            </div>
          </div>
        </div>

        <div className="text-sm text-gray-500 text-center">
          点击"打印"按钮使用浏览器打印功能
        </div>
      </div>

      {/* 打印样式 */}
      <style>{`
        @media print {
          .print\:hidden { display: none !important; }
          body * { visibility: hidden; }
          .print-content, .print-content * { visibility: visible; }
          .print-content { position: absolute; left: 0; top: 0; }
        }
      `}</style>
    </UnifiedModal>
  );
}
```
