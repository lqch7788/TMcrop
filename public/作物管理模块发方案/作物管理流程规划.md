## 一、项目背景与问题分析

### 1.1 当前系统问题

经过深入代码分析，发现以下关键问题：

#### 问题1：类型定义重复冲突
| 类型 | 位置1 | 位置2 | 冲突 |
|------|--------|--------|------|
| `SourceOrigin` | `crop.ts` 行21-28 (枚举) | `crop.ts` 行31-40 (类型联合) | 值完全不同 |
| `SeedlingStatus` | `crop.ts` (in_progress) | `crop/seedling.ts` (seedling) | 值不同 |
| `PlantingStatus` | `crop.ts` (growing) | `crop/planting.ts` (harvesting) | 值不同 |

#### 问题2：数据源不一致
```
cropData.ts (简化数据 - 仅6个作物)
    ├── 番茄、黄瓜、生菜、茄子、辣椒、豆角
    │
    └── 问题：与 produceCodeRule.ts 不同步！

    ├── 蔬菜类(PD): 叶菜类、瓜菜类、茄果类、根茎类、豆类、葱蒜类
    ├── 水果类(FR): 浆果类、核果类、仁果类、柑橘类、热带水果
    ├── 粮食类(GR): 稻谷类、小麦类、玉米类
    └── ...更多类别

    ├── cropToCategoryMap - 独立维护的映射表
    └── varietyToCodesMap - 品种代码映射
```

### 2.2 设计原则

1. **单一数据源**：所有作物数据统一存储在作物品种库
2. **固定编码**：每个作物品种有固定不变的9位编码
3. **动态获取**：所有模块从品种库API获取数据，不维护本地映射
4. **快速扩展**：支持在任意业务场景中快速新增品种

## 三、数据模型设计

### 3.1 作物品种库 (CropVariety) - 核心实体

```typescript
/**
 * 作物品种库 - 系统数据基础表
 * 编码规则：类别(2位) + 类型(2位) + 品种(2位) + 流水号(3位) = 9位
 * 示例：PD0301001 = 蔬菜类-茄果类-番茄-第1个编码
 */
interface CropVariety {
  id: string;                    // 唯一ID (CV + 时间戳)
  cropCode: string;             // 作物编码(9位) - 全系统唯一，固定不变
  categoryCode: string;         // 类别代码 (如 'PD')
  categoryName: string;          // 类别名称 (如 '蔬菜类')
  typeCode: string;             // 类型代码 (如 '03')
  typeName: string;             // 类型名称 (如 '茄果类')
  varietyCode: string;           // 品种代码 (如 '01')
  varietyName: string;           // 品种名称 (如 '番茄')
  alias?: string[];              // 别名/俗名 (如 ['西红柿', '洋柿子'])
  growthCycle?: number;          // 生长周期(天)
  suitableTemp?: string;         // 适宜温度范围
  targetYield?: number;          // 目标产量(kg/亩)
  yieldUnit?: string;            // 产量单位
  status: 'active' | 'inactive'; // 状态：启用/停用
  remarks?: string;              // 备注说明
  createTime: string;            // 创建时间
  updateTime: string;            // 更新时间
}
```

## 四、服务层设计

### 4.1 作物品种库服务 (cropVarietyService.ts)

```typescript
// ============================================
// 作物品种库服务 - 单一数据源
// ============================================

const STORAGE_KEY = 'crop_varieties';

// 默认品种数据（从 produceCodeRule 导入）
const defaultVarieties: CropVariety[] = [
  // 蔬菜类 - 茄果类 - 番茄
  {
    id: 'CV001',
    cropCode: 'PD0301001',
    categoryCode: 'PD',
    categoryName: '蔬菜类',
    typeCode: '03',
    typeName: '茄果类',
    varietyCode: '01',
    varietyName: '番茄',
    alias: ['西红柿', '洋柿子'],
    growthCycle: 120,
    status: 'active',
    createTime: '2026-01-01 00:00:00',
    updateTime: '2026-01-01 00:00:00'
  },
  // ... 更多品种从 produceCodeRule 动态导入
];

/**
 * 初始化品种库（首次加载时从 produceCodeRule 导入）
 */
export function initVarieties(): CropVariety[];

/**
 * 获取所有品种
 */
export function getAllVarieties(): CropVariety[];

/**
 * 按类别获取品种
 */
export function getVarietiesByCategory(categoryCode: string): CropVariety[];

/**
 * 根据ID获取品种
 */
export function getVarietyById(id: string): CropVariety | undefined;

/**
 * 根据编码获取品种
 */
export function getVarietyByCode(cropCode: string): CropVariety | undefined;

/**
 * 搜索品种（按编码、名称、别名搜索）
 */
export function searchVarieties(keyword: string): CropVariety[];

/**
 * 新增品种
 */
export function addVariety(variety: Omit<CropVariety, 'id' | 'createTime' | 'updateTime'>): CropVariety;

/**
 * 更新品种（仅允许更新 alias, growthCycle, status, remarks）
 */
export function updateVariety(id: string, updates: Partial<CropVariety>): CropVariety | null;

/**
 * 删除品种
 */
export function deleteVariety(id: string): boolean;

/**
 * 生成新的作物编码
 * @param categoryCode 类别代码
 * @param typeCode 类型代码
 * @param varietyCode 品种代码
 * @returns 新的9位作物编码
 */
export function generateCropCode(categoryCode: string, typeCode: string, varietyCode: string): string;

/**
 * 获取品种选择器的选项格式
 */
export function getVarietyOptions(): Array<{
  value: string;      // cropCode
  label: string;      // varietyName
  category: string;   // categoryName
  type: string;       // typeName
  alias?: string[];
}>;
```
