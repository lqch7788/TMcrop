// src/utils/migrationValidator.ts
// 迁移校验服务

interface ValidationResult {
  passed: boolean;
  message: string;
  details?: any;
}

interface ValidationReport {
  total: number;
  passed: number;
  failed: number;
  results: ValidationResult[];
}

class MigrationValidator {
  
  // 校验记录数一致性
  async validateRecordCount(
    tableName: string,
    localStorageKey: string
  ): Promise<ValidationResult> {
    const localData = localStorage.getItem(localStorageKey);
    const localCount = localData ? JSON.parse(localData).length : 0;
    
    try {
      const response = await fetch(`/api/${tableName}/count`);
      const result = await response.json();
      const sqliteCount = result.count || 0;
      
      const passed = localCount === sqliteCount;
      return {
        passed,
        message: passed 
          ? `✅ ${tableName}: ${localCount} 条` 
          : `❌ ${tableName}: localStorage=${localCount}, SQLite=${sqliteCount}`,
        details: { localCount, sqliteCount }
      };
    } catch (e) {
      return { passed: false, message: `API 调用失败: ${e}` };
    }
  }
  
  // 校验外键完整性
  async validateForeignKeys(): Promise<ValidationResult> {
    try {
      const response = await fetch('/api/integrity/foreign-keys');
      const result = await response.json();
      const issues = result.issues || [];
      
      return {
        passed: issues.length === 0,
        message: issues.length === 0 
          ? '✅ 外键关联完整' 
          : `❌ 有 ${issues.length} 个外键问题`,
        details: { issues }
      };
    } catch (e) {
      return { passed: false, message: `校验失败: ${e}` };
    }
  }
  
  // 执行全部校验
  async validateAll(): Promise<ValidationReport> {
    const results: ValidationResult[] = [];
    
    const checks = [
      { table: 'crop-variety', key: 'crop_varieties' },
      { table: 'seed-source', key: 'crop_seed_sources' },
      { table: 'seedling', key: 'crop_seedlings' },
      { table: 'planting', key: 'crop_plantings' },
      { table: 'harvest', key: 'harvest_records' },
      { table: 'crop-order', key: 'crop_orders' },
      { table: 'crop-instance', key: 'crop_instances' },
    ];
    
    for (const check of checks) {
      results.push(await this.validateRecordCount(check.table, check.key));
    }
    
    results.push(await this.validateForeignKeys());
    
    const passedCount = results.filter(r => r.passed).length;
    return {
      total: results.length,
      passed: passedCount,
      failed: results.length - passedCount,
      results
    };
  }
}

export const migrationValidator = new MigrationValidator();
```

// src/utils/migrationExecutor.ts
// 迁移执行器

interface MigrationResult {
  success: boolean;
  message: string;
  details?: {
    successCount: number;
    failCount: number;
    sqliteCount: number;
    errors: string[];
  };
}

class MigrationExecutor {
  private validator = migrationValidator;
  
  // 执行完整迁移
  async execute(): Promise<void> {
    console.log('='.repeat(50));
    console.log('🚀 数据迁移开始');
    console.log('='.repeat(50));
    
    // Batch 1: 品种库
    console.log('\n📦 Batch 1: 品种库...');
    await this.migrateBatch('crop_varieties', 'crop-variety');
    
    // Batch 2: 种源
    console.log('\n📦 Batch 2: 种源...');
    await this.migrateBatch('crop_seed_sources', 'seed-source');
    
    // Batch 3: 育苗
    console.log('\n📦 Batch 3: 育苗...');
    await this.migrateBatch('crop_seedlings', 'seedling');
    
    // Batch 4: 种植
    console.log('\n📦 Batch 4: 种植...');
    await this.migrateBatch('crop_plantings', 'planting');
    
    // Batch 5: 采收
    console.log('\n📦 Batch 5: 采收...');
    await this.migrateBatch('harvest_records', 'harvest');
    
    // Batch 6: 订单
    console.log('\n📦 Batch 6: 订单...');
    await this.migrateBatch('crop_orders', 'crop-order');
    
    // Batch 7: 实例
    console.log('\n📦 Batch 7: 实例...');
    await this.migrateBatch('crop_instances', 'crop-instance');
    
    // 校验
    console.log('\n🔍 执行校验...');
    const report = await this.validator.validateAll();
    
    if (report.failed > 0) {
      console.error('\n❌ 校验未通过:');
      report.results.filter(r => !r.passed).forEach(r => console.error(`  ${r.message}`));
      return;
    }
    
    console.log('\n✅ 迁移完成！所有数据已迁移到 SQLite');
    console.log('='.repeat(50));
  }
  
  // 迁移单个批次
  private async migrateBatch(storageKey: string, apiEndpoint: string): Promise<MigrationResult> {
    const localData = localStorage.getItem(storageKey);
    if (!localData) {
      return { success: true, message: '无数据' };
    }
    
    const items = JSON.parse(localData);
    let successCount = 0;
    let failCount = 0;
    const errors: string[] = [];
    
    for (const item of items) {
      try {
        const response = await fetch(`/api/${apiEndpoint}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(item)
        });
        
        if (response.ok) successCount++;
        else {
          failCount++;
          errors.push(`${item.id}: ${response.statusText}`);
        }
      } catch (e) {
        failCount++;
        errors.push(`${item.id}: ${e}`);
      }
    }
    
    console.log(`  成功 ${successCount}, 失败 ${failCount}`);
    
    return {
      success: failCount === 0,
      message: `成功 ${successCount}, 失败 ${failCount}`,
      details: { successCount, failCount, errors: errors.slice(0, 5) }
    };
  }
}

export const migrationExecutor = new MigrationExecutor();
```

function gradualMiddleware(req, res, next) {
  const config = gradualConfig.getConfig();
  
  req.gradualConfig = config;
  
  const userId = req.user?.id;
  const random = Math.random();
  
  if (config.stage.name === 'shadow') {
    req.useSQLite = false;
    req.dualWrite = true;
  } else if (config.stage.name === 'trial') {
    req.useSQLite = gradualConfig.isUserInTrial(userId) && random < config.stage.readRatio;
    req.dualWrite = true;
  } else {
    req.useSQLite = random < config.stage.readRatio;
    req.dualWrite = config.features.dualWrite;
  }
  
  console.log(`[Gradual] Stage: ${config.stage.name}, useSQLite: ${req.useSQLite}, user: ${userId}`);
  
  next();
}

### 7.1 备份服务

```javascript
// server/services/backupService.js
const fs = require('fs');
const path = require('path');

const DB_PATH = path.join(__dirname, '../data/TMcrop.db');
const BACKUP_DIR = path.join(__dirname, '../backup');

class BackupService {
  constructor() {
    this.initDirs();
  }

  initDirs() {
    ['hourly', 'daily', 'release', 'realtime'].forEach(dir => {
      fs.mkdirSync(path.join(BACKUP_DIR, dir), { recursive: true });
    });
  }

  // L1: 实时备份
  async realtimeBackup() {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupFile = path.join(BACKUP_DIR, 'realtime', `realtime_${timestamp}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // 保留最近 100 个
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'realtime'))
        .filter(f => f.startsWith('realtime_'))
        .sort()
        .reverse();
      
      if (files.length > 100) {
        files.slice(100).forEach(f => {
          fs.unlinkSync(path.join(BACKUP_DIR, 'realtime', f));
        });
      }
      
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L2: 小时备份
  async hourlyBackup() {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupFile = path.join(BACKUP_DIR, 'hourly', `hourly_${timestamp}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // 保留 72 小时
      const cutoff = Date.now() - (72 * 60 * 60 * 1000);
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'hourly'));
      
      for (const file of files) {
        const filePath = path.join(BACKUP_DIR, 'hourly', file);
        const stats = fs.statSync(filePath);
        if (stats.mtime.getTime() < cutoff) {
          fs.unlinkSync(filePath);
        }
      }
      
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L3: 日备份
  async dailyBackup() {
    const day = new Date().toISOString().split('T')[0].replace(/-/g, '');
    const backupFile = path.join(BACKUP_DIR, 'daily', `daily_${day}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L4: 版本备份
  async releaseBackup(version) {
    const day = new Date().toISOString().split('T')[0].replace(/-/g, '');
    const backupFile = path.join(BACKUP_DIR, 'release', `release_${version}_${day}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // 保留最近 10 个
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'release'))
        .filter(f => f.startsWith('release_'))
        .sort()
        .reverse();
      
      if (files.length > 10) {
        files.slice(10).forEach(f => {
          fs.unlinkSync(path.join(BACKUP_DIR, 'release', f));
        });
      }
      
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // 启动定时任务
  startSchedules() {
    // 小时备份
    setInterval(() => {
      this.hourlyBackup().then(r => console.log('[Backup] Hourly:', r.success ? 'OK' : 'FAIL'));
    }, 60 * 60 * 1000);

    // 日备份（午夜）
    const now = new Date();
    const midnight = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1, 0, 0, 0);
    const msUntilMidnight = midnight - now;
    
    setTimeout(() => {
      this.dailyBackup().then(r => console.log('[Backup] Daily:', r.success ? 'OK' : 'FAIL'));
      setInterval(() => {
        this.dailyBackup().then(r => console.log('[Backup] Daily:', r.success ? 'OK' : 'FAIL'));
      }, 24 * 60 * 60 * 1000);
    }, msUntilMidnight);

    console.log('[Backup] Schedules started');
  }
}

module.exports = { BackupService };
```

#!/bin/bash
# server/scripts/backup.sh
# 用法: ./backup.sh [realtime|hourly|daily|release]

DB_PATH="data/TMcrop.db"
BACKUP_DIR="backup"
DATE=$(date +%Y%m%d_%H%M%S)

mkdir -p "$BACKUP_DIR"/{hourly,daily,release,realtime}

case "${1:-daily}" in
  realtime)
    cp "$DB_PATH" "$BACKUP_DIR/realtime/realtime_${DATE}.db"
    ls -t "$BACKUP_DIR/realtime/realtime_"*.db | tail -n +101 | xargs -r rm
    echo "[Backup] Realtime: $BACKUP_DIR/realtime/realtime_${DATE}.db"
    ;;
  hourly)
    cp "$DB_PATH" "$BACKUP_DIR/hourly/hourly_${DATE}.db"
    find "$BACKUP_DIR/hourly" -name "hourly_*.db" -mtime +3 -delete
    echo "[Backup] Hourly: $BACKUP_DIR/hourly/hourly_${DATE}.db"
    ;;
  daily)
    cp "$DB_PATH" "$BACKUP_DIR/daily/daily_${DATE}.db"
    find "$BACKUP_DIR/daily" -name "daily_*.db" -mtime +30 -delete
    echo "[Backup] Daily: $BACKUP_DIR/daily/daily_${DATE}.db"
    ;;
  release)
    VERSION="${2:-v$(date +%s)}"
    cp "$DB_PATH" "$BACKUP_DIR/release/release_${VERSION}_${DATE}.db"
    ls -t "$BACKUP_DIR/release/release_"*.db | tail -n +11 | xargs -r rm
    echo "[Backup] Release: $BACKUP_DIR/release/release_${VERSION}_${DATE}.db"
    ;;
esac

## 八、自动回滚检测器（V8.0 实现）

### 8.1 回滚检测器

```javascript
// server/middleware/rollbackDetector.js
class RollbackDetector {
  constructor() {
    this.metrics = {
      apiErrors: [],
      dbFailures: [],
      dataCorruptions: [],
    };
    this.thresholds = {
      apiErrorRate: { threshold: 0.05, window: 5 * 60 * 1000 },
      dbFailureCount: { threshold: 3, window: 60 * 1000 },
      dataCorruption: { threshold: 1, window: 1000 },
    };
    this.isHealthy = true;
  }

  recordApiError(statusCode, path) {
    this.metrics.apiErrors.push({ time: Date.now(), statusCode, path });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  recordDbFailure(error) {
    this.metrics.dbFailures.push({ time: Date.now(), error: error.message });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  recordDataCorruption(details) {
    this.metrics.dataCorruptions.push({ time: Date.now(), details });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  cleanOldMetrics() {
    const now = Date.now();
    this.metrics.apiErrors = this.metrics.apiErrors.filter(
      m => now - m.time < this.thresholds.apiErrorRate.window
    );
    this.metrics.dbFailures = this.metrics.dbFailures.filter(
      m => now - m.time < this.thresholds.dbFailureCount.window
    );
    this.metrics.dataCorruptions = this.metrics.dataCorruptions.filter(
      m => now - m.time < this.thresholds.dataCorruption.window
    );
  }

  checkHealth() {
    // API 错误率检查
    const apiErrorRate = this.metrics.apiErrors.length / 100;
    if (apiErrorRate > this.thresholds.apiErrorRate.threshold) {
      this.triggerRollback('API_ERROR_RATE', `错误率 ${(apiErrorRate * 100).toFixed(1)}% 超过阈值`);
      return;
    }
    
    // DB 失败次数检查
    if (this.metrics.dbFailures.length >= this.thresholds.dbFailureCount.threshold) {
      this.triggerRollback('DB_FAILURE', `${this.metrics.dbFailures.length} 次 DB 失败`);
      return;
    }
    
    // 数据损坏检查
    if (this.metrics.dataCorruptions.length >= this.thresholds.dataCorruption.threshold) {
      this.triggerRollback('DATA_CORRUPTION', '检测到数据损坏');
      return;
    }
  }

  triggerRollback(reason, details) {
    console.error(`🚨 [Rollback] 触发自动回滚: ${reason} - ${details}`);
    
    // 切换到 localStorage 模式
    // 通知前端切换
    this.isHealthy = false;
    
    // 发送告警
    console.error('🚨 [Alert] 需要人工介入检查系统');
  }

  getHealthStatus() {
    return {
      isHealthy: this.isHealthy,
      metrics: {
        apiErrors: this.metrics.apiErrors.length,
        dbFailures: this.metrics.dbFailures.length,
        dataCorruptions: this.metrics.dataCorruptions.length,
      }
    };
  }
}

module.exports = { RollbackDetector };
```
