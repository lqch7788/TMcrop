## 二、完整数据库Schema（56张表）

### 2.1 基础数据层（10张表）

```sql
-- ============================================
-- 基础数据层：用户/员工/部门/岗位/角色/权限/公司/基地/大棚
-- ============================================

-- 1. 用户表（登录账号）
CREATE TABLE IF NOT EXISTS users (
  id TEXT PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  password_hash TEXT,
  real_name TEXT NOT NULL,
  department_id TEXT,
  position TEXT,
  role TEXT DEFAULT 'user',
  status TEXT DEFAULT 'active',
  phone TEXT,
  email TEXT,
  avatar TEXT,
  last_login_time TEXT,
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (department_id) REFERENCES departments(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_users_username ON users(username);
CREATE INDEX IF NOT EXISTS idx_users_status ON users(status);

-- 2. 员工表（HR档案）
CREATE TABLE IF NOT EXISTS staff (
  id TEXT PRIMARY KEY,
  staff_code TEXT UNIQUE NOT NULL,
  staff_name TEXT NOT NULL,
  department_id TEXT,
  position_id TEXT,
  join_date TEXT,
  employee_type TEXT DEFAULT 'permanent',
  id_card TEXT,
  phone TEXT,
  emergency_contact TEXT,
  emergency_phone TEXT,
  bank_account TEXT,
  hourly_rate REAL DEFAULT 25,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (department_id) REFERENCES departments(id) ON DELETE SET NULL,
  FOREIGN KEY (position_id) REFERENCES positions(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_staff_department ON staff(department_id);
CREATE INDEX IF NOT EXISTS idx_staff_status ON staff(status);

-- 3. 部门表
CREATE TABLE IF NOT EXISTS departments (
  id TEXT PRIMARY KEY,
  department_code TEXT UNIQUE NOT NULL,
  department_name TEXT NOT NULL,
  parent_id TEXT,
  manager_id TEXT,
  status TEXT DEFAULT 'active',
  sort_order INTEGER DEFAULT 0,
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (parent_id) REFERENCES departments(id) ON DELETE SET NULL,
  FOREIGN KEY (manager_id) REFERENCES staff(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_departments_parent ON departments(parent_id);

-- 4. 岗位表
CREATE TABLE IF NOT EXISTS positions (
  id TEXT PRIMARY KEY,
  position_code TEXT UNIQUE NOT NULL,
  position_name TEXT NOT NULL,
  department_id TEXT,
  level TEXT,
  description TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (department_id) REFERENCES departments(id) ON DELETE SET NULL
);

-- 5. 角色表
CREATE TABLE IF NOT EXISTS roles (
  id TEXT PRIMARY KEY,
  role_code TEXT UNIQUE NOT NULL,
  role_name TEXT NOT NULL,
  description TEXT,
  company_group_id TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (company_group_id) REFERENCES company_groups(id) ON DELETE SET NULL
);

-- 6. 权限表
CREATE TABLE IF NOT EXISTS permissions (
  id TEXT PRIMARY KEY,
  permission_code TEXT UNIQUE NOT NULL,
  permission_name TEXT NOT NULL,
  resource_type TEXT,
  action_type TEXT,
  description TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL
);

-- 7. 角色权限关联表
CREATE TABLE IF NOT EXISTS role_permissions (
  id TEXT PRIMARY KEY,
  role_id TEXT NOT NULL,
  permission_id TEXT NOT NULL,
  create_time TEXT NOT NULL,
  FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE,
  FOREIGN KEY (permission_id) REFERENCES permissions(id) ON DELETE CASCADE,
  UNIQUE(role_id, permission_id)
);

-- 8. 公司分组表
CREATE TABLE IF NOT EXISTS company_groups (
  id TEXT PRIMARY KEY,
  group_code TEXT UNIQUE NOT NULL,
  group_name TEXT NOT NULL,
  description TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT
);

-- 9. 基地表
CREATE TABLE IF NOT EXISTS bases (
  id TEXT PRIMARY KEY,
  base_code TEXT UNIQUE NOT NULL,
  base_name TEXT NOT NULL,
  location TEXT,
  area_size REAL,
  company_group_id TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (company_group_id) REFERENCES company_groups(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_bases_company ON bases(company_group_id);

-- 10. 大棚/温室表
CREATE TABLE IF NOT EXISTS greenhouses (
  id TEXT PRIMARY KEY,
  greenhouse_code TEXT UNIQUE NOT NULL,
  greenhouse_name TEXT NOT NULL,
  base_id TEXT,
  area_name TEXT,
  area_size REAL,
  greenhouse_type TEXT,
  status TEXT DEFAULT 'active',
  create_time TEXT NOT NULL,
  update_time TEXT,
  FOREIGN KEY (base_id) REFERENCES bases(id) ON DELETE SET NULL
);

CREATE INDEX IF NOT EXISTS idx_greenhouses_base ON greenhouses(base_id);
```

## 三、审批联动触发器（完整SQL）

### 3.1 审批通过后自动更新业务状态

```sql
-- ============================================
-- 审批联动触发器：审批通过后自动执行业务动作
-- ============================================

-- 触发器1：采购审批通过后更新采购单状态
CREATE TRIGGER IF NOT EXISTS trg_approval_passed_purchase
AFTER UPDATE ON approvals
WHEN NEW.status = 'approved' AND OLD.status = 'pending' AND NEW.approval_type = 'purchase'
BEGIN
  UPDATE purchase_orders 
  SET status = 'approved', 
      approved_by = NEW.current_node_id,
      approved_at = datetime('now')
  WHERE id = NEW.business_id;
  
  -- 记录审计日志
  INSERT INTO audit_logs (
    id, operation_type, resource_type, resource_id, operation,
    operator_id, operation_detail, result, create_time
  ) VALUES (
    lower(hex(randomblob(16))), 'approval', 'purchase_order', NEW.business_id,
    'approve', NEW.applicant_id, '采购审批通过，单号：' || NEW.approval_code,
    'success', datetime('now')
  );
END;

-- 触发器2：请假审批通过后更新考勤状态
CREATE TRIGGER IF NOT EXISTS trg_approval_passed_leave
AFTER UPDATE ON approvals
WHEN NEW.status = 'approved' AND OLD.status = 'pending' AND NEW.approval_type = 'leave'
BEGIN
  UPDATE leave_records 
  SET status = 'approved', 
      approver_id = NEW.current_node_id,
      update_time = datetime('now')
  WHERE id = NEW.business_id;
  
  -- 同步更新考勤记录为请假状态
  UPDATE attendance_records 
  SET status = 'leave',
      leave_type = (SELECT leave_type FROM leave_records WHERE id = NEW.business_id),
      update_time = datetime('now')
  WHERE staff_id = NEW.applicant_id 
    AND record_date BETWEEN 
      (SELECT start_date FROM leave_records WHERE id = NEW.business_id)
      AND (SELECT end_date FROM leave_records WHERE id = NEW.business_id);
END;

-- 触发器3：加班审批通过后更新考勤状态
CREATE TRIGGER IF NOT EXISTS trg_approval_passed_overtime
AFTER UPDATE ON approvals
WHEN NEW.status = 'approved' AND OLD.status = 'pending' AND NEW.approval_type = 'overtime'
BEGIN
  UPDATE overtime_records 
  SET status = 'approved',
      update_time = datetime('now')
  WHERE id = NEW.business_id;
  
  -- 同步更新考勤记录的加班时长
  UPDATE attendance_records 
  SET overtime_hours = (SELECT hours FROM overtime_records WHERE id = NEW.business_id),
      update_time = datetime('now')
  WHERE staff_id = NEW.applicant_id 
    AND record_date = (SELECT overtime_date FROM overtime_records WHERE id = NEW.business_id);
END;

-- 触发器4：薪资调整审批通过后更新薪资记录
CREATE TRIGGER IF NOT EXISTS trg_approval_passed_salary
AFTER UPDATE ON approvals
WHEN NEW.status = 'approved' AND OLD.status = 'pending' AND NEW.approval_type = 'salary_adjustment'
BEGIN
  UPDATE salary_adjustments 
  SET status = 'approved',
      effective_date = date('now'),
      update_time = datetime('now')
  WHERE id = NEW.business_id;
  
  -- 同步更新员工表薪资
  UPDATE staff 
  SET hourly_rate = (SELECT new_salary FROM salary_adjustments WHERE id = NEW.business_id)
  WHERE id = (SELECT staff_id FROM salary_adjustments WHERE id = NEW.business_id);
END;

-- 触发器5：离职审批通过后更新员工状态
CREATE TRIGGER IF NOT EXISTS trg_approval_passed_resignation
AFTER UPDATE ON approvals
WHEN NEW.status = 'approved' AND OLD.status = 'pending' AND NEW.approval_type = 'resignation'
BEGIN
  UPDATE resignations 
  SET status = 'approved',
      update_time = datetime('now')
  WHERE id = NEW.business_id;
  
  -- 同步更新员工状态为离职
  UPDATE staff 
  SET status = 'resigned',
      update_time = datetime('now')
  WHERE id = (SELECT staff_id FROM resignations WHERE id = NEW.business_id);
  
  -- 同步更新用户状态为禁用
  UPDATE users 
  SET status = 'inactive',
      update_time = datetime('now')
  WHERE id = (SELECT id FROM users WHERE real_name = (SELECT staff_name FROM staff WHERE id = (SELECT staff_id FROM resignations WHERE id = NEW.business_id)));
END;

-- 触发器6：通用审批驳回处理
CREATE TRIGGER IF NOT EXISTS trg_approval_rejected
AFTER UPDATE ON approvals
WHEN NEW.status = 'rejected' AND OLD.status = 'pending'
BEGIN
  -- 更新对应业务表状态为驳回
  CASE NEW.approval_type
    WHEN 'purchase' THEN
      UPDATE purchase_orders SET status = 'rejected', update_time = datetime('now') WHERE id = NEW.business_id;
    WHEN 'leave' THEN
      UPDATE leave_records SET status = 'rejected', update_time = datetime('now') WHERE id = NEW.business_id;
    WHEN 'overtime' THEN
      UPDATE overtime_records SET status = 'rejected', update_time = datetime('now') WHERE id = NEW.business_id;
    WHEN 'salary_adjustment' THEN
      UPDATE salary_adjustments SET status = 'rejected', update_time = datetime('now') WHERE id = NEW.business_id;
    WHEN 'resignation' THEN
      UPDATE resignations SET status = 'rejected', update_time = datetime('now') WHERE id = NEW.business_id;
  END;
  
  -- 记录审计日志
  INSERT INTO audit_logs (
    id, operation_type, resource_type, resource_id, operation,
    operator_id, operation_detail, result, create_time
  ) VALUES (
    lower(hex(randomblob(16))), 'approval', NEW.approval_type, NEW.business_id,
    'reject', NEW.current_node_id, '审批被驳回，单号：' || NEW.approval_code,
    'success', datetime('now')
  );
END;
```

```javascript
  if (!obj || typeof obj !== 'object') return obj;
```

```bash
#!/bin/bash
# ============================================
# TMcrop 4级备份脚本
# Usage: ./backup.sh [level]
# Levels: realtime | hourly | daily | release
# ============================================

DB_PATH="data/TMcrop.db"
BACKUP_DIR="backup"
DATE=$(date +%Y%m%d_%H%M%S)
DAY=$(date +%Y%m%d)

# Create backup directories
mkdir -p "$BACKUP_DIR/hourly"
mkdir -p "$BACKUP_DIR/daily"
mkdir -p "$BACKUP_DIR/release"
mkdir -p "$BACKUP_DIR/realtime"

case "${1:-daily}" in
  realtime)
    # L1: 实时备份（每次写入后触发，保留最近100次）
    BACKUP_FILE="$BACKUP_DIR/realtime/realtime_${DATE}.db"
    cp "$DB_PATH" "$BACKUP_FILE"
    # Keep only last 100 realtime backups
    ls -t "$BACKUP_DIR/realtime/realtime_*.db" | tail -n +101 | xargs -r rm
    echo "[Backup] Realtime backup: $BACKUP_FILE"
    ;;
    
  hourly)
    # L2: 小时备份（每小时执行，保留72小时）
    BACKUP_FILE="$BACKUP_DIR/hourly/hourly_${DATE}.db"
    cp "$DB_PATH" "$BACKUP_FILE"
    # Keep only last 72 hourly backups (3 days)
    find "$BACKUP_DIR/hourly" -name "hourly_*.db" -mtime +3 -delete
    echo "[Backup] Hourly backup: $BACKUP_FILE"
    ;;
    
  daily)
    # L3: 日备份（每日0点执行，保留30天）
    BACKUP_FILE="$BACKUP_DIR/daily/daily_${DAY}.db"
    cp "$DB_PATH" "$BACKUP_FILE"
    # Compress daily backup
    gzip -f "$BACKUP_FILE"
    # Keep only last 30 daily backups
    find "$BACKUP_DIR/daily" -name "daily_*.db.gz" -mtime +30 -delete
    echo "[Backup] Daily backup: ${BACKUP_FILE}.gz"
    ;;
    
  release)
    # L4: 版本备份（每次部署执行，保留10个版本）
    VERSION="${2:-v$(date +%s)}"
    BACKUP_FILE="$BACKUP_DIR/release/release_${VERSION}_${DAY}.db"
    cp "$DB_PATH" "$BACKUP_FILE"
    gzip -f "$BACKUP_FILE"
    # Keep only last 10 release backups
    ls -t "$BACKUP_DIR/release/release_*.db.gz" | tail -n +11 | xargs -r rm
    echo "[Backup] Release backup: ${BACKUP_FILE}.gz"
    ;;
    
  *)
    echo "Usage: $0 [realtime|hourly|daily|release] [version]"
    exit 1
    ;;
esac
```

### 5.2 Node.js定时备份服务

```javascript
/**
 * Backup service with 4-level backup strategy
 * backend/services/backupService.js
 */
const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');
const util = require('util');
const execPromise = util.promisify(exec);

const DB_PATH = path.join(__dirname, '../data/TMcrop.db');
const BACKUP_DIR = path.join(__dirname, '../backup');

class BackupService {
  constructor() {
    this.intervals = {};
    this.initDirs();
  }

  initDirs() {
    ['hourly', 'daily', 'release', 'realtime'].forEach(dir => {
      fs.mkdirSync(path.join(BACKUP_DIR, dir), { recursive: true });
    });
  }

  // L1: Realtime backup (triggered after every write operation)
  async realtimeBackup() {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupFile = path.join(BACKUP_DIR, 'realtime', `realtime_${timestamp}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // Keep only last 100 realtime backups
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'realtime'))
        .filter(f => f.startsWith('realtime_'))
        .sort()
        .reverse();
      
      if (files.length > 100) {
        files.slice(100).forEach(f => {
          fs.unlinkSync(path.join(BACKUP_DIR, 'realtime', f));
        });
      }
      
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L2: Hourly backup
  async hourlyBackup() {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
    const backupFile = path.join(BACKUP_DIR, 'hourly', `hourly_${timestamp}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // Clean up old hourly backups (keep 72 hours)
      const cutoff = Date.now() - (72 * 60 * 60 * 1000);
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'hourly'));
      
      for (const file of files) {
        const filePath = path.join(BACKUP_DIR, 'hourly', file);
        const stats = fs.statSync(filePath);
        if (stats.mtime.getTime() < cutoff) {
          fs.unlinkSync(filePath);
        }
      }
      
      return { success: true, file: backupFile };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L3: Daily backup
  async dailyBackup() {
    const day = new Date().toISOString().split('T')[0].replace(/-/g, '');
    const backupFile = path.join(BACKUP_DIR, 'daily', `daily_${day}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      
      // Compress
      await execPromise(`gzip -f "${backupFile}"`);
      
      // Clean up old daily backups (keep 30 days)
      const cutoff = Date.now() - (30 * 24 * 60 * 60 * 1000);
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'daily'));
      
      for (const file of files) {
        if (!file.endsWith('.gz')) continue;
        const filePath = path.join(BACKUP_DIR, 'daily', file);
        const stats = fs.statSync(filePath);
        if (stats.mtime.getTime() < cutoff) {
          fs.unlinkSync(filePath);
        }
      }
      
      return { success: true, file: `${backupFile}.gz` };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // L4: Release backup
  async releaseBackup(version) {
    const day = new Date().toISOString().split('T')[0].replace(/-/g, '');
    const backupFile = path.join(BACKUP_DIR, 'release', `release_${version}_${day}.db`);
    
    try {
      fs.copyFileSync(DB_PATH, backupFile);
      await execPromise(`gzip -f "${backupFile}"`);
      
      // Keep only last 10 release backups
      const files = fs.readdirSync(path.join(BACKUP_DIR, 'release'))
        .filter(f => f.startsWith('release_'))
        .sort()
        .reverse();
      
      if (files.length > 10) {
        files.slice(10).forEach(f => {
          fs.unlinkSync(path.join(BACKUP_DIR, 'release', f));
        });
      }
      
      return { success: true, file: `${backupFile}.gz` };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  // Start automatic backup schedules
  startSchedules() {
    // Hourly backup every hour
    this.intervals.hourly = setInterval(() => {
      this.hourlyBackup().then(r => console.log('[Backup] Hourly:', r.success ? 'OK' : 'FAIL'));
    }, 60 * 60 * 1000);

    // Daily backup at midnight
    const now = new Date();
    const midnight = new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1, 0, 0, 0);
    const msUntilMidnight = midnight - now;
    
    setTimeout(() => {
      this.dailyBackup().then(r => console.log('[Backup] Daily:', r.success ? 'OK' : 'FAIL'));
      this.intervals.daily = setInterval(() => {
        this.dailyBackup().then(r => console.log('[Backup] Daily:', r.success ? 'OK' : 'FAIL'));
      }, 24 * 60 * 60 * 1000);
    }, msUntilMidnight);

    console.log('[Backup] Schedules started. Hourly + Daily (at midnight).');
  }

  stopSchedules() {
    Object.values(this.intervals).forEach(clearInterval);
    console.log('[Backup] Schedules stopped.');
  }
}

module.exports = { BackupService };
```

```javascript
/**
 * Automatic rollback detector
 * backend/middleware/rollbackDetector.js

class RollbackDetector {
  constructor() {
    this.metrics = {
      apiErrors: [],
      dbFailures: [],
      dataCorruptions: [],
    };
    this.thresholds = {
      apiErrorRate: { threshold: 0.05, window: 5 * 60 * 1000 }, // 5% error rate in 5min
      dbFailureCount: { threshold: 3, window: 60 * 1000 }, // 3 failures in 1min
      dataCorruption: { threshold: 1, window: 1000 }, // Any corruption signal
    };
    this.isHealthy = true;
    this.lastAlert = null;
  }

  recordApiError(statusCode, path) {
    this.metrics.apiErrors.push({ time: Date.now(), statusCode, path });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  recordDbFailure(error) {
    this.metrics.dbFailures.push({ time: Date.now(), error: error.message });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  recordDataCorruption(details) {
    this.metrics.dataCorruptions.push({ time: Date.now(), details });
    this.cleanOldMetrics();
    this.checkHealth();
  }

  cleanOldMetrics() {
    const now = Date.now();
    this.metrics.apiErrors = this.metrics.apiErrors.filter(m => now - m.time < this.thresholds.apiErrorRate.window);
    this.metrics.dbFailures = this.metrics.dbFailures.filter(m => now - m.time < this.thresholds.dbFailure.window);
    this.metrics.dataCorruptions = this.metrics.dataCorruptions.filter(m => now - m.time < this.thresholds.dataCorruption.window);
  }

  checkHealth() {
    const now = Date.now();
    
    // Check API error rate
    const apiErrorRate = this.metrics.apiErrors.length / 100; // Approximate based on recent requests
    if (apiErrorRate > this.thresholds.apiErrorRate.threshold) {
      this.triggerRollback('API_ERROR_RATE', `Error rate ${(apiErrorRate * 100).toFixed(1)}% exceeds threshold`);
      return;
    }
    
    // Check DB failures
    if (this.metrics.dbFailures.length >= this.thresholds.dbFailureCount.threshold) {
      this.triggerRollback('DB_FAILURE', `${this.metrics.dbFailures.length} DB failures in 1 minute`);
      return;
    }
    
    // Check data corruption
    if (this.metrics.dataCorruptions.length >= this.thresholds.dataCorruption.threshold) {
      this.triggerRollback('DATA_CORRUPTION', 'Data corruption signal detected');
      return;
    }
  }

  triggerRollback(reason, details) {
    if (!this.isHealthy) return; // Already in rollback state
    
    this.isHealthy = false;
    this.lastAlert = { time: Date.now(), reason, details };
    
    console.error(`[Rollback] TRIGGERED: ${reason} - ${details}`);
    
    // 1. Stop accepting new writes
    global.ROLLBACK_MODE = true;
    
    // 2. Alert administrators
    this.sendAlert(reason, details);
    
    // 3. Switch to read-only mode
    // 4. Prepare for manual intervention or automatic recovery
  }

  sendAlert(reason, details) {
    // Send notification via email/SMS/WeChat
    const alert = {
      type: 'ROLLBACK_ALERT',
      reason,
      details,
      timestamp: new Date().toISOString(),
      severity: 'CRITICAL',
    };
    
    // Write to audit log
    console.error('[Alert]', JSON.stringify(alert));
    
    // TODO: Integrate with notification service
  }

  recover() {
    this.isHealthy = true;
    global.ROLLBACK_MODE = false;
    this.metrics = { apiErrors: [], dbFailures: [], dataCorruptions: [] };
    console.log('[Rollback] Recovered. Normal operation resumed.');
  }
}

// Global instance
const rollbackDetector = new RollbackDetector();

// Express middleware
function rollbackMiddleware(req, res, next) {
  if (global.ROLLBACK_MODE && req.method !== 'GET') {
    return res.status(503).json({
      error: 'Service in rollback mode',
      message: 'Write operations temporarily disabled due to system health concerns.',
    });
  }
  
  // Record response for health monitoring
  res.on('finish', () => {
    if (res.statusCode >= 500) {
      rollbackDetector.recordApiError(res.statusCode, req.path);
    }
  });
  
  next();
}
```

### 8.1 Circuit Breaker

```javascript
/**
 * Circuit Breaker pattern implementation
 * backend/utils/circuitBreaker.js
 */
class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.recoveryTimeout = options.recoveryTimeout || 30000;
    this.halfOpenMaxCalls = options.halfOpenMaxCalls || 3;
    
    this.state = 'CLOSED'; // CLOSED | OPEN | HALF_OPEN
    this.failureCount = 0;
    this.successCount = 0;
    this.lastFailureTime = null;
    this.halfOpenCalls = 0;
    
    this.onStateChange = options.onStateChange || (() => {});
  }

  async call(fn, ...args) {
    if (this.state === 'OPEN') {
      if (Date.now() - this.lastFailureTime > this.recoveryTimeout) {
        this.transitionTo('HALF_OPEN');
      } else {
        throw new CircuitBreakerOpenError('Circuit breaker is OPEN');
      }
    }

    if (this.state === 'HALF_OPEN') {
      if (this.halfOpenCalls >= this.halfOpenMaxCalls) {
        throw new CircuitBreakerOpenError('Circuit breaker HALF_OPEN limit reached');
      }
      this.halfOpenCalls++;
    }

    try {
      const result = await fn(...args);
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  onSuccess() {
    this.failureCount = 0;
    
    if (this.state === 'HALF_OPEN') {
      this.successCount++;
      if (this.successCount >= this.halfOpenMaxCalls) {
        this.transitionTo('CLOSED');
      }
    }
  }

  onFailure() {
    this.failureCount++;
    this.lastFailureTime = Date.now();
    
    if (this.state === 'HALF_OPEN') {
      this.transitionTo('OPEN');
      return;
    }
    
    if (this.failureCount >= this.failureThreshold) {
      this.transitionTo('OPEN');
    }
  }

  transitionTo(newState) {
    const oldState = this.state;
    this.state = newState;
    
    if (newState === 'CLOSED') {
      this.failureCount = 0;
      this.successCount = 0;
      this.halfOpenCalls = 0;
    } else if (newState === 'HALF_OPEN') {
      this.halfOpenCalls = 0;
      this.successCount = 0;
    }
    
    this.onStateChange(oldState, newState);
    console.log(`[CircuitBreaker] ${oldState} → ${newState}`);
  }

  getState() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      successCount: this.successCount,
      lastFailureTime: this.lastFailureTime,
    };
  }
}

class CircuitBreakerOpenError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CircuitBreakerOpenError';
  }
}

// Factory for different services
const breakers = {};

function getBreaker(serviceName, options = {}) {
  if (!breakers[serviceName]) {
    breakers[serviceName] = new CircuitBreaker({
      ...options,
      onStateChange: (oldState, newState) => {
        console.log(`[CircuitBreaker:${serviceName}] ${oldState} → ${newState}`);
      },
    });
  }
```

```javascript
/**
 * Data integrity checker
 * backend/scripts/dataIntegrityCheck.js

const Database = require('better-sqlite3');

class DataIntegrityChecker {
  constructor(dbPath) {
    this.db = new Database(dbPath);
    this.issues = [];
  }

  checkAll() {
    console.log('[Integrity] Starting full data integrity check...');
    
    this.checkForeignKeys();
    this.checkEnumValues();
    this.checkDateLogic();
    this.checkLinkIntegrity();
    this.checkApprovalSync();
    this.checkInventorySync();
    
    console.log(`[Integrity] Check complete. Found ${this.issues.length} issues.`);
    
    if (this.issues.length > 0) {
      console.log('[Integrity] Issues:');
      this.issues.forEach((issue, i) => {
        console.log(`  ${i + 1}. [${issue.severity}] ${issue.check}: ${issue.message}`);
      });
    }
    
    return {
      passed: this.issues.length === 0,
      issueCount: this.issues.length,
      issues: this.issues,
    };
  }

  checkForeignKeys() {
    // Check seed_sources → suppliers
    const orphanSources = this.db.prepare(`
      SELECT s.id, s.seed_code, s.supplier_id 
      FROM seed_sources s 
      LEFT JOIN suppliers sup ON sup.id = s.supplier_id 
      WHERE s.supplier_id IS NOT NULL AND sup.id IS NULL
    `).all();
    
    if (orphanSources.length > 0) {
      this.issues.push({
        check: 'FK_seed_sources_suppliers',
        severity: 'HIGH',
        message: `${orphanSources.length} seed_sources have invalid supplier_id`,
        details: orphanSources.map(r => r.seed_code),
      });
    }

    // Check seedlings → seed_sources
    const orphanSeedlings = this.db.prepare(`
      SELECT sl.id, sl.batch_code, sl.source_id 
      FROM seedlings sl 
      LEFT JOIN seed_sources s ON s.id = sl.source_id 
      WHERE s.id IS NULL
    `).all();
    
    if (orphanSeedlings.length > 0) {
      this.issues.push({
        check: 'FK_seedlings_seed_sources',
        severity: 'HIGH',
        message: `${orphanSeedlings.length} seedlings have invalid source_id`,
        details: orphanSeedlings.map(r => r.batch_code),
      });
    }

    // Check plantings → seedlings
    const orphanPlantings = this.db.prepare(`
      SELECT p.id, p.plant_code, p.seedling_id 
      FROM plantings p 
      LEFT JOIN seedlings sl ON sl.id = p.seedling_id 
      WHERE p.seedling_id IS NOT NULL AND sl.id IS NULL
    `).all();
    
    if (orphanPlantings.length > 0) {
      this.issues.push({
        check: 'FK_plantings_seedlings',
        severity: 'HIGH',
        message: `${orphanPlantings.length} plantings have invalid seedling_id`,
        details: orphanPlantings.map(r => r.plant_code),
      });
    }

    // Check staff → departments
    const orphanStaff = this.db.prepare(`
      SELECT st.id, st.staff_name, st.department_id 
      FROM staff st 
      LEFT JOIN departments d ON d.id = st.department_id 
      WHERE st.department_id IS NOT NULL AND d.id IS NULL
    `).all();
    
    if (orphanStaff.length > 0) {
      this.issues.push({
        check: 'FK_staff_departments',
        severity: 'MEDIUM',
        message: `${orphanStaff.length} staff have invalid department_id`,
        details: orphanStaff.map(r => r.staff_name),
      });
    }
  }

  checkEnumValues() {
    // Check status values
    const validStatuses = ['active', 'inactive', 'pending', 'approved', 'rejected', 'completed', 'cancelled', 'draft'];
    
    const tables = ['seed_sources', 'seedlings', 'plantings', 'harvest_records', 'crop_orders', 'staff', 'approvals'];
    
    for (const table of tables) {
      try {
        const invalid = this.db.prepare(`
          SELECT id, status FROM ${table} 
          WHERE status IS NOT NULL AND status NOT IN (${validStatuses.map(() => '?').join(',')})
        `).all(...validStatuses);
        
        if (invalid.length > 0) {
          this.issues.push({
            check: `ENUM_${table}_status`,
            severity: 'MEDIUM',
            message: `${invalid.length} records in ${table} have invalid status`,
            details: invalid.map(r => `${r.id}=${r.status}`),
          });
        }
      } catch (e) {
        // Table might not have status column
      }
    }
  }

  checkDateLogic() {
    // Check planting dates
    const invalidPlantingDates = this.db.prepare(`
      SELECT id, plant_date, expected_harvest_date 
      FROM plantings 
      WHERE plant_date > expected_harvest_date
    `).all();
    
    if (invalidPlantingDates.length > 0) {
      this.issues.push({
        check: 'DATE_planting_harvest',
        severity: 'MEDIUM',
        message: `${invalidPlantingDates.length} plantings have plant_date > expected_harvest_date`,
      });
    }

    // Check leave dates
    const invalidLeaveDates = this.db.prepare(`
      SELECT id, start_date, end_date 
      FROM leave_records 
      WHERE start_date > end_date
    `).all();
    
    if (invalidLeaveDates.length > 0) {
      this.issues.push({
        check: 'DATE_leave_range',
        severity: 'MEDIUM',
        message: `${invalidLeaveDates.length} leave records have start_date > end_date`,
      });
    }
  }

  checkLinkIntegrity() {
    // Check seed source → planting → harvest chain
    const brokenChain = this.db.prepare(`
      SELECT ss.id as source_id, ss.seed_code,
             sl.id as seedling_id, sl.batch_code,
             p.id as planting_id, p.plant_code,
             h.id as harvest_id, h.harvest_code
      FROM seed_sources ss
      LEFT JOIN seedlings sl ON sl.source_id = ss.id
      LEFT JOIN plantings p ON p.source_id = ss.id
      LEFT JOIN harvest_records h ON h.source_id = ss.id
      WHERE ss.status = 'active' 
        AND sl.id IS NULL 
        AND p.id IS NULL 
        AND h.id IS NULL
    `).all();
    
    if (brokenChain.length > 0) {
      this.issues.push({
        check: 'CHAIN_seed_to_harvest',
        severity: 'LOW',
        message: `${brokenChain.length} active seed_sources have no linked seedlings/plantings/harvests`,
        details: brokenChain.map(r => r.seed_code),
      });
    }

    // Check harvest → inventory
    const orphanHarvests = this.db.prepare(`
      SELECT h.id, h.harvest_code 
      FROM harvest_records h 
      LEFT JOIN produce_inventories i ON i.harvest_record_id = h.id 
      WHERE h.status = 'completed' AND i.id IS NULL
    `).all();
    
    if (orphanHarvests.length > 0) {
      this.issues.push({
        check: 'CHAIN_harvest_inventory',
        severity: 'LOW',
        message: `${orphanHarvests.length} completed harvests not in inventory`,
        details: orphanHarvests.map(r => r.harvest_code),
      });
    }
  }

  checkApprovalSync() {
    // Check approval → business status sync
    const unsyncedApprovals = this.db.prepare(`
      SELECT a.id, a.approval_code, a.approval_type, a.status as approval_status,
             l.status as business_status
      FROM approvals a
      JOIN leave_records l ON l.id = a.business_id
      WHERE a.approval_type = 'leave'
        AND a.status = 'approved'
        AND l.status != 'approved'
    `).all();
    
    if (unsyncedApprovals.length > 0) {
      this.issues.push({
        check: 'SYNC_approval_leave',
        severity: 'HIGH',
        message: `${unsyncedApprovals.length} approved leave approvals not synced to leave_records`,
      });
    }
  }

  checkInventorySync() {
    // Check inventory quantities match harvest totals
    const inventoryMismatch = this.db.prepare(`
      SELECT i.crop_name, i.quantity as inventory_qty,
             SUM(h.harvest_quantity) as harvest_total
      FROM produce_inventories i
      LEFT JOIN harvest_records h ON h.crop_name = i.crop_name AND h.status = 'completed'
      GROUP BY i.crop_name
      HAVING ABS(i.quantity - COALESCE(SUM(h.harvest_quantity), 0)) > 0.01
    `).all();
    
    if (inventoryMismatch.length > 0) {
      this.issues.push({
        check: 'SYNC_inventory_harvest',
        severity: 'MEDIUM',
        message: `${inventoryMismatch.length} inventory items don't match harvest totals`,
        details: inventoryMismatch.map(r => `${r.crop_name}: inv=${r.inventory_qty} harvest=${r.harvest_total}`),
      });
    }
  }

  autoFix() {
    console.log('[Integrity] Attempting auto-fix...');
    let fixed = 0;

    // Fix orphan staff departments
    const orphanStaff = this.db.prepare(`
      SELECT st.id FROM staff st 
      LEFT JOIN departments d ON d.id = st.department_id 
      WHERE st.department_id IS NOT NULL AND d.id IS NULL
    `).all();
    
    for (const staff of orphanStaff) {
      this.db.prepare("UPDATE staff SET department_id = NULL WHERE id = ?").run(staff.id);
      fixed++;
    }

    console.log(`[Integrity] Auto-fixed ${fixed} issues.`);
    return fixed;
  }

  close() {
    this.db.close();
  }
}

// Run if called directly
if (require.main === module) {
  const checker = new DataIntegrityChecker('data/TMcrop.db');
  const result = checker.checkAll();
  
  if (!result.passed) {
    const fixed = checker.autoFix();
    console.log(`[Integrity] Auto-fixed ${fixed} issues.`);
    
    // Re-check
    const recheck = checker.checkAll();
    console.log(`[Integrity] After fix: ${recheck.issueCount} issues remaining.`);
  }
  
  checker.close();
}

module.exports = { DataIntegrityChecker };
```

```typescript
  async linkInstances(orderId: string, instanceIds: string[]) {
    const order = (cropOrders || []).find(o => o.id === orderId);
    if (order) {
      order.instanceIds = [...(order.instanceIds || []), ...instanceIds];
      order.updateTime = now();
      return true;
    }
    return false;
  },
  async unlinkInstances(orderId: string, instanceIds: string[]) {
    const order = (cropOrders || []).find(o => o.id === orderId);
    if (order && order.instanceIds) {
      order.instanceIds = order.instanceIds.filter((id: string) => !instanceIds.includes(id));
      order.updateTime = now();
      return true;
    }
    return false;
  },
  async updateOrderStatus(id: string, status: string) {
    const order = (cropOrders || []).find(o => o.id === id);
    if (order) {
      order.status = status;
      order.updateTime = now();
      return true;
    }
    return false;
  },
  async getOrderDetail(id: string) {
    const order = (cropOrders || []).find(o => o.id === id);
    if (order) {
      return { ...order, instances: order.instanceIds || [] };
    }
    return null;
  },
  async resetOrders() {
    if (cropOrders) cropOrders.length = 0;
  }
};
```
