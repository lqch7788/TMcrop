### 1.3 数据孤岛问题汇总

| 问题 | 现状 | 理想状态 |
|------|------|---------|
| 种源→育苗 | `sourceId` 硬编码引用，无级联更新 | 外键关联 + 触发器同步 |
| 育苗→种植 | `sourceId` 引用，但数量不自动同步 | 事务自动扣减/增加 |
| 种植→采收 | `batchCode` 字符串匹配，易出错 | 外键关联到 plantings.id |
| 采收→库存 | `warehouseId` 写入，但库存表独立 | 采收自动触发入库 |
| 实例→全链路 | 各模块操作不更新实例数量 | 所有操作围绕实例自动同步 |
| 订单→实例 | `instanceIds` 数组不同步 | 自动创建/绑定实例 |
| 人员→操作 | `createBy` 是字符串"李明辉" | 关联 staff.id，支持历史追溯 |
| 基地→种植 | `areaName` 是字符串"一棚>01区" | 关联 greenhouse/plant_area 表 |
| 审批→业务 | 审批与业务数据分离 | 审批关联具体业务记录 |
| 考勤→人员 | 考勤独立，无人员档案关联 | 关联 staff 表的完整在职状态 |

### 2.5 迁移原则

1. **数据完整性**：所有历史数据必须完整迁移，不丢失任何业务记录
2. **关联可追溯**：通过外键关联实现端到端追溯
3. **灰度迁移**：分模块渐进迁移，每步可回滚
4. **业务连续性**：迁移期间系统不中断，所有功能正常可用
5. **零风险回滚**：任何阶段失败可一键回退到迁移前状态

```typescript
    flowType: ApprovalFlowType;

    content: Record<string, any>;
    attachments?: string[];
    ccIds?: string[];
  }): Promise<Approval> {
    // 1. 获取流程定义
    const flow = await this.getActiveFlow(params.flowType);
    if (!flow) {
      throw new Error(`未找到审批流程: ${params.flowType}`);
    }
    
    // 2. 创建审批记录
    const approval = await db.biz_approvals.create({
      approval_code: this.generateCode(params.flowType),
      flow_id: flow.id,
      flow_type: params.flowType,
      title: params.title,
      applicant_id: currentUser.staffId,
      applicant_name: currentUser.name,
      current_node_id: flow.start_node_id,
      content: JSON.stringify(params.content),
      attachments: JSON.stringify(params.attachments || []),
      cc_ids: JSON.stringify(params.ccIds || []),
      status: 'pending',
      apply_date: new Date().toISOString(),
    });
    
    // 3. 创建首节点记录
    await this.createNodeRecord(approval.id, flow.start_node_id, 'submit');
    
    // 4. 通知第一审批人
    await this.notifyApprovers(approval, flow.start_node_id);
    
    return approval;
  }
  
  // 审批操作
  async approve(params: {
    approvalId: string;
    action: ApprovalAction;
    comment?: string;
    transferToId?: string;
  }): Promise<Approval> {
    const approval = await db.biz_approvals.get(params.approvalId);
    const currentNode = await this.getCurrentNode(approval);
    
    // 验证审批权限
    if (!this.canApprove(approval, currentUser)) {
      throw new Error('您没有权限审批此申请');
    }
    
    // 执行审批动作
    switch (params.action) {
      case ApprovalAction.APPROVE:
        return await this.handleApprove(approval, currentNode, params.comment);
      case ApprovalAction.REJECT:
        return await this.handleReject(approval, currentNode, params.comment);
      case ApprovalAction.TRANSFER:
        return await this.handleTransfer(approval, currentNode, params.transferToId, params.comment);
      case ApprovalAction.RECALL:
        return await this.handleRecall(approval);
      default:
        throw new Error('无效的审批动作');
    }
  }
  
  // 处理同意
  private async handleApprove(approval: Approval, node: ApprovalNode, comment?: string): Promise<Approval> {
    // 1. 记录审批
    await this.createNodeRecord(approval.id, node.id, 'approve', comment);
    
    // 2. 查找下一节点
    const nextNode = await this.getNextNode(node);
    
    if (!nextNode || nextNode.node_type === 'end') {
      // 审批完成
      await db.biz_approvals.update(approval.id, {
        status: 'approved',
        approve_date: new Date().toISOString(),
        current_node_id: null,
      });
      
      // 执行审批通过后续操作
      await this.executePostApproval(approval);
    } else {
      // 进入下一审批节点
      await db.biz_approvals.update(approval.id, {
        current_node_id: nextNode.id,
        current_node_name: nextNode.node_name,
      });
      
      // 通知下一审批人
      await this.notifyApprovers(approval, nextNode);
    }
    
    return await db.biz_approvals.get(approval.id);
  }
  
  // 执行审批通过后续操作
  private async executePostApproval(approval: Approval): Promise<void> {
    switch (approval.flow_type) {
      case ApprovalFlowType.LEAVE:
        // 更新请假记录状态
        await db.biz_leave_records.update(approval.related_record_id, {
          status: 'approved',
          approver_id: currentUser.staffId,
        });
        break;
        
      case ApprovalFlowType.PURCHASE:
        // 更新采购申请状态
        await db.biz_material_requisitions.update(approval.related_record_id, {
          status: 'approved',
          approver_id: currentUser.staffId,
        });
        break;
        
      // ... 其他类型处理
    }
    
    // 记录审计日志
    await this.logApprovalAction(approval, 'approved');
  }
}
```

```typescript
  }
```

### 7.2.2 数据一致性校验核心实现

```typescript
// 数据一致性校验服务
class DataIntegrityValidator {
  
  // 迁移前基线校验
  async captureBaseline(): Promise<DataBaseline> {
    const baseline = {
      timestamp: new Date().toISOString(),
      localStorage: {} as Record<string, { count: number; checksum: string }>,
      indexedDB: {} as Record<string, { count: number; checksum: string }>
    };
    
    // LocalStorage 基线
    const lsKeys = [
      'crop_orders', 'crop_instances', 'crop_seed_sources', 
      'crop_seedlings', 'crop_plantings', 'harvest_records',
      'yuanxingtu_tasks', 'yuanxingtu_inspections'
    ];
    
    for (const key of lsKeys) {
      const data = localStorage.getItem(key);
      if (data) {
        const parsed = JSON.parse(data);
        baseline.localStorage[key] = {
          count: Array.isArray(parsed) ? parsed.length : 0,
          checksum: this.simpleChecksum(parsed)
        };
      }
    }
    
    return baseline;
  }
  
  // 简单校验和计算
  private simpleChecksum(data: any): string {
    const str = JSON.stringify(data);
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash.toString(16);
  }
  
  // 迁移后对比校验
  async validateMigration(baseline: DataBaseline): Promise<ValidationResult> {
    const result: ValidationResult = {
      passed: true,
      differences: []
    };
    
    // 对比 LocalStorage
    for (const [key, original] of Object.entries(baseline.localStorage)) {
      const current = localStorage.getItem(key);
      const currentData = current ? JSON.parse(current) : [];
      const currentCount = Array.isArray(currentData) ? currentData.length : 0;
      
      if (currentCount !== original.count) {
        result.passed = false;
        result.differences.push({
          source: 'localStorage',
          key,
          type: 'count_mismatch',
          expected: original.count,
          actual: currentCount
        });
      }
    }
    
    // 对比 IndexedDB/SQLite
    const db = new Dexie('CropManagementDB');
    for (const [table, original] of Object.entries(baseline.indexedDB)) {
      try {
        const records = await db.table(table).toArray();
        if (records.length !== original.count) {
          result.passed = false;
          result.differences.push({
            source: 'indexedDB',
            table,
            type: 'count_mismatch',
            expected: original.count,
            actual: records.length
          });
        }
      } catch (e) {
        console.warn(`校验表 ${table} 失败:`, e);
      }
    }
    
    return result;
  }
  
  // 关键关联校验
  async validateRelations(): Promise<RelationValidationResult> {
    const issues: RelationIssue[] = [];
    
    // 1. 订单-实例关系
    const orders = JSON.parse(localStorage.getItem('crop_orders') || '[]');
    const instances = JSON.parse(localStorage.getItem('crop_instances') || '[]');
    
    for (const order of orders) {
      if (order.instanceIds) {
        for (const instId of order.instanceIds) {
          const exists = instances.find(i => i.id === instId);
          if (!exists) {
            issues.push({
              type: 'orphaned_reference',
              from: `orders:${order.id}`,
              to: `instances:${instId}`,
              description: `订单 ${order.orderCode} 引用了不存在的实例 ${instId}`
            });
          }
        }
      }
    }
    
    // 2. 实例-种源关系
    const seedSources = JSON.parse(localStorage.getItem('crop_seed_sources') || '[]');
    for (const inst of instances) {
      if (inst.seedSourceIds) {
        for (const ssId of inst.seedSourceIds) {
          const exists = seedSources.find(s => s.id === ssId);
          if (!exists) {
            issues.push({
              type: 'orphaned_reference',
              from: `instances:${inst.id}`,
              to: `seedSources:${ssId}`,
              description: `实例 ${inst.instanceCode} 引用了不存在的种源 ${ssId}`
            });
          }
        }
      }
    }
    
    // 3. 育苗-种源关系
    const seedlings = JSON.parse(localStorage.getItem('crop_seedlings') || '[]');
    for (const sd of seedlings) {
      if (sd.sourceId) {
        const exists = seedSources.find(s => s.id === sd.sourceId);
        if (!exists) {
          issues.push({
            type: 'orphaned_reference',
            from: `seedlings:${sd.id}`,
            to: `seedSources:${sd.sourceId}`,
            description: `育苗 ${sd.seedlingCode} 引用了不存在的种源 ${sd.sourceId}`
          });
        }
      }
    }
    
    return { valid: issues.length === 0, issues };
  }
  
  // 外键完整性校验（SQLite 迁移后）
  async validateForeignKeys(sqliteDb: any): Promise<FKValidationResult> {
    const issues: FKIssue[] = [];
    
    // 校验种源-供应商
    const orphanSuppliers = await sqliteDb.exec(`
      SELECT id FROM biz_seed_sources 
      WHERE supplier_id IS NOT NULL 
      AND supplier_id NOT IN (SELECT id FROM biz_suppliers)
    `);
    if (orphanSuppliers.length > 0) {
      issues.push({
        table: 'biz_seed_sources',
        field: 'supplier_id',
        orphanedCount: orphanSuppliers.length,
        description: '种源引用了不存在的供应商'
      });
    }
    
    // 校验种源-创建人
    const orphanCreators = await sqliteDb.exec(`
      SELECT id FROM biz_seed_sources 
      WHERE create_by IS NOT NULL 
      AND create_by NOT IN (SELECT id FROM biz_staff)
    `);
    if (orphanCreators.length > 0) {
      issues.push({
        table: 'biz_seed_sources',
        field: 'create_by',
        orphanedCount: orphanCreators.length,
        description: '种源引用了不存在的人员'
      });
    }
    
    // 校验育苗-种源
    const orphanSeedSources = await sqliteDb.exec(`
      SELECT id FROM biz_seedlings 
      WHERE source_id IS NOT NULL 
      AND source_id NOT IN (SELECT id FROM biz_seed_sources)
    `);
    if (orphanSeedSources.length > 0) {
      issues.push({
        table: 'biz_seedlings',
        field: 'source_id',
        orphanedCount: orphanSeedSources.length,
        description: '育苗引用了不存在的种源'
      });
    }
    
    return { valid: issues.length === 0, issues };
  }
  
  // 数量守恒校验
  async validateQuantityConservation(sqliteDb: any): Promise<QCValidationResult> {
    const errors: QCError[] = [];
    
    const instances = await sqliteDb.exec('SELECT * FROM biz_crop_instances');
    for (const inst of instances) {
      const expected = inst.initial_quantity 
        - inst.planted_quantity 
        - inst.harvested_quantity 
        - inst.loss_quantity;
      
      if (inst.current_quantity !== expected) {
        errors.push({
          instanceCode: inst.instance_code,
          expected: expected,
          actual: inst.current_quantity,
          description: `实例 ${inst.instance_code} 数量不守恒`
        });
      }
    }
    
    return { valid: errors.length === 0, errors };
  }
}
```

```typescript
      this.performHealthCheck();
```

```typescript
// 性能监控指标
interface PerformanceMetric {
  operation: string;
  duration: number;
  timestamp: number;
  success: boolean;
}

// 性能监控服务
class PerformanceMonitor {
  
  private slowQueryThreshold = 300;
  private metrics: PerformanceMetric[] = [];
  private maxMetrics = 1000;
  
  // 记录指标
  private recordMetric(metric: PerformanceMetric): void {
    this.metrics.push(metric);
    if (this.metrics.length > this.maxMetrics) {
      this.metrics.shift();
    }
  }
  
  // 测量操作性能
  async measure<T>(operation: string, fn: () => Promise<T>): Promise<T> {
    const start = performance.now();
    let success = true;
    let result: T;
    
    try {
      result = await fn();
    } catch (error) {
      success = false;
      throw error;
    } finally {
      const duration = performance.now() - start;
      this.recordMetric({
        operation,
        duration,
        timestamp: Date.now(),
        success
      });
      
      if (duration > this.slowQueryThreshold) {
        this.sendAlert({ type: 'slow_query', operation, duration });
      }
    }
    
    return result!;
  }
  
  // 发送告警
  private sendAlert(alert: { type: string; operation: string; duration: number }): void {
    console.warn(`🚨 [性能告警] ${alert.operation} 耗时 ${alert.duration.toFixed(2)}ms`);
    // 可扩展：发送到监控系统
  }
  
  // 生成报告
  generateReport(): {
    avgLatency: number;
    maxLatency: number;
    minLatency: number;
    totalOperations: number;
    successRate: number;
    slowQueries: PerformanceMetric[];
  } {
    const successfulMetrics = this.metrics.filter(m => m.success);
    const slowQueries = this.metrics.filter(m => m.duration > this.slowQueryThreshold);
    
    if (successfulMetrics.length === 0) {
      return {
        avgLatency: 0,
        maxLatency: 0,
        minLatency: 0,
        totalOperations: 0,
        successRate: 100,
        slowQueries: []
      };
    }
    
    const latencies = successfulMetrics.map(m => m.duration);
    
    return {
      avgLatency: latencies.reduce((a, b) => a + b, 0) / latencies.length,
      maxLatency: Math.max(...latencies),
      minLatency: Math.min(...latencies),
      totalOperations: this.metrics.length,
      successRate: (successfulMetrics.length / this.metrics.length) * 100,
      slowQueries
    };
  }
  
  // 获取最近N条记录
  getRecentMetrics(count: number = 10): PerformanceMetric[] {
    return this.metrics.slice(-count);
  }
  
  // 清空指标
  clearMetrics(): void {
    this.metrics = [];
  }
}

class MigrationCircuitBreaker {
  private failureCount = 0;
  private successCount = 0;
  private threshold = 10;        // 连续失败10次触发熔断
  private recoveryThreshold = 5; // 连续成功5次恢复
  private state: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED';
  private halfOpenSuccess = 0;
  
  // 记录成功
  recordSuccess(): void {
    this.failureCount = 0;
    this.successCount++;
    if (this.state === 'HALF_OPEN') {
      this.halfOpenSuccess++;
      if (this.halfOpenSuccess >= this.recoveryThreshold) {
        this.state = 'CLOSED';
        console.log('🔄 熔断器恢复：连续成功5次');
      }
    }
  }
  
  // 记录失败
  recordFailure(): void {
    this.failureCount++;
    this.successCount = 0;
    if (this.failureCount >= this.threshold) {
      this.state = 'OPEN';
      console.log('🚨 熔断器触发：连续失败10次，暂停迁移');
    }
  }
  
  // 检查是否可以继续
  canProceed(): boolean {
    return this.state !== 'OPEN';
  }
  
  // 获取当前状态
  getState(): string {
    return this.state;
  }
  
  // 重置熔断器
  reset(): void {
    this.failureCount = 0;
    this.successCount = 0;
    this.halfOpenSuccess = 0;
    this.state = 'CLOSED';
  }
}

// 迁移健康监控
const MigrationMonitor = {
  // 监控指标

    this.metrics.avgLatency = (this.metrics.avgLatency + latency) / 2;

    });

    if (this.metrics.errorRate > 0.01) {
      this.triggerAlert(`错误率超过1%: ${(this.metrics.errorRate * 100).toFixed(2)}%`);

    };
  }
```

```typescript
        create_time: new Date().toISOString(),
        update_time: new Date().toISOString()
      });
    }
  }
}
```
