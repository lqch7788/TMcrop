### 2.1 最高原则：零数据丢失

```

│                      零数据丢失原则                                      │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  1. 迁移前必须备份                                                      │
│     └─ 导出全部 localStorage 到文件                                      │
│                                                                          │
│  2. 迁移中必须校验                                                      │
│     └─ 每批次迁移后对比记录数                                            │
│                                                                          │
│  3. 切换后必须验证                                                      │
│     └─ API 读取数据与 localStorage 一致                                 │
│                                                                          │
│  4. 失败后必须回滚                                                      │
│     └─ 一键切回 localStorage                                            │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘
```

### 2.2 迁移策略：渐进式三步走

| 阶段 | 操作 | 风险 | 数据源 |
|------|------|------|--------|
| **Step 1** | 数据导入 | 低（只写） | localStorage → SQLite |
| **Step 2** | 灰度验证 | 中（读切换） | SQLite（读） + localStorage（写） |
| **Step 3** | 全量切换 | 低 | SQLite（读写） |

```typescript
// 迁移校验服务
class MigrationValidator {
  
  // 校验项 1: 记录数一致
  async validateRecordCount(
    tableName: string, 
    localStorageKey: string
  ): Promise<ValidationResult> {
    const localData = localStorage.getItem(localStorageKey);
    const localCount = localData ? JSON.parse(localData).length : 0;
    
    let sqliteCount = 0;
    try {
      const response = await fetch(`/api/${tableName}/count`);
      const result = await response.json();
      sqliteCount = result.count;
    } catch (e) {
      return { passed: false, message: `API 调用失败: ${e}` };
    }
    
    const passed = localCount === sqliteCount;
    return {
      passed,
      message: passed 
        ? `✅ ${tableName}: ${localCount} 条` 
        : `❌ ${tableName}: localStorage=${localCount}, SQLite=${sqliteCount}`,
      details: { localCount, sqliteCount }
    };
  }
  
  // 校验项 2: 关键字段完整性
  async validateFieldIntegrity(
    tableName: string, 
    requiredFields: string[]
  ): Promise<ValidationResult> {
    const response = await fetch(`/api/${tableName}/integrity-check`);
    const result = await response.json();
    
    const issues = result.issues || [];
    const passed = issues.length === 0;
    
    return {
      passed,
      message: passed 
        ? `✅ ${tableName} 字段完整性通过` 
        : `❌ ${tableName} 有 ${issues.length} 个字段问题`,
      details: { issues }
    };
  }
  
  // 校验项 3: 外键关联
  async validateForeignKeys(): Promise<ValidationResult> {
    const response = await fetch('/api/integrity/foreign-keys');
    const result = await response.json();
    
    const issues = result.issues || [];
    const passed = issues.length === 0;
    
    return {
      passed,
      message: passed 
        ? `✅ 外键关联完整` 
        : `❌ 有 ${issues.length} 个外键问题`,
      details: { issues }
    };
  }
  
  // 执行全部校验
  async validateAll(): Promise<ValidationReport> {
    const results: ValidationResult[] = [];
    
    // 1. 校验记录数
    const recordCountChecks = [
      { table: 'crop_varieties', key: 'crop_varieties' },
      { table: 'seed_sources', key: 'crop_seed_sources' },
      { table: 'seedlings', key: 'crop_seedlings' },
      { table: 'plantings', key: 'crop_plantings' },
      { table: 'harvest_records', key: 'harvest_records' },
      { table: 'crop_orders', key: 'crop_orders' },
      { table: 'crop_instances', key: 'crop_instances' },
    ];
    
    for (const check of recordCountChecks) {
      results.push(await this.validateRecordCount(check.table, check.key));
    }
    
    // 2. 校验外键
    results.push(await this.validateForeignKeys());
    
    const passedCount = results.filter(r => r.passed).length;
    return {
      total: results.length,
      passed: passedCount,
      failed: results.length - passedCount,
      results
    };
  }
}
```

```typescript
async function migrateCropVarieties(): Promise<MigrationResult> {
  console.log('📦 开始迁移品种库...');
  
  // 1. 读取 localStorage
  const localData = localStorage.getItem('crop_varieties');
  if (!localData) {
    return { success: true, message: '无品种库数据' };
  }
  
  const varieties = JSON.parse(localData);
  console.log(`  发现 ${varieties.length} 条品种记录`);
  
  // 2. 逐条迁移
  let successCount = 0;
  let failCount = 0;
  const errors: string[] = [];
  
  for (const variety of varieties) {
    try {
      const response = await fetch('/api/crop-varieties', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(variety)
      });
      
      if (response.ok) {
        successCount++;
      } else {
        failCount++;
        errors.push(`ID ${variety.id} 失败: ${response.statusText}`);
      }
    } catch (e) {
      failCount++;
      errors.push(`ID ${variety.id} 异常: ${e}`);
    }
  }
  
  // 3. 校验
  const verifyResponse = await fetch('/api/crop-varieties/count');
  const { count } = await verifyResponse.json();
  
  console.log(`✅ 品种库迁移完成: 成功 ${successCount}, 失败 ${failCount}`);
  console.log(`  SQLite 验证: ${count} 条`);
  
  if (failCount > 0) {
    console.log('❌ 错误列表:', errors.slice(0, 5));
  }
  
  return {
    success: failCount === 0,
    message: `成功 ${successCount}, 失败 ${failCount}`,
    details: { successCount, failCount, sqliteCount: count, errors: errors.slice(0, 10) }
  };
}
```

### 4.4 Phase 2: 灰度验证

#### 灰度切换机制

```typescript
// 灰度控制器
class GrayScaleController {
  // 灰度比例配置
  private grayScalePercent = 0; // 默认 0%
  
  // 判断当前用户是否走 API
  shouldUseAPI(): boolean {
    // 如果强制使用 API
    if (localStorage.getItem('force_api_mode') === 'true') {
      return true;
    }
    
    // 如果强制使用 localStorage
    if (localStorage.getItem('force_local_mode') === 'true') {
      return false;
    }
    
    // 基于灰度比例
    const userId = localStorage.getItem('userId') || '';
    const hash = this.hashCode(userId);
    const percent = hash % 100;
    
    return percent < this.grayScalePercent;
  }
  
  // 切换灰度比例
  setGrayScalePercent(percent: number): void {
    this.grayScalePercent = percent;
    localStorage.setItem('gray_scale_percent', String(percent));
    console.log(`🔄 灰度比例已调整为 ${percent}%`);
  }
  
  // 一键切换到全量 API
  switchToFullAPI(): void {
    localStorage.setItem('force_api_mode', 'true');
    localStorage.removeItem('force_local_mode');
    console.log('✅ 已切换到全量 API 模式');
  }
  
  // 一键回滚到 localStorage
  switchToLocalStorage(): void {
    localStorage.setItem('force_local_mode', 'true');
    localStorage.removeItem('force_api_mode');
    console.log('⚠️ 已回滚到 localStorage 模式');
  }
  
  private hashCode(str: string): number {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return Math.abs(hash);
  }
}
```

```typescript
async function executeFullSwitch(): Promise<void> {

  const validator = new MigrationValidator();
```

## 六、迁移执行脚本

### 6.1 一键迁移脚本

```typescript
// 迁移执行器
class MigrationExecutor {
  private validator = new MigrationValidator();
  
  // 执行完整迁移
  async execute(): Promise<void> {
    console.log('='.repeat(60));
    console.log('🚀 数据迁移开始');
    console.log('='.repeat(60));
    
    // Phase 0: 准备
    console.log('\n📋 Phase 0: 准备工作');
    await this.phase0();
    
    // Phase 1: 迁移
    console.log('\n📦 Phase 1: 数据迁移');
    await this.phase1();
    
    // Phase 2: 校验
    console.log('\n🔍 Phase 2: 校验');
    const report = await this.validator.validateAll();
    
    if (!report.passed) {
      console.error('\n❌ 校验未通过，迁移失败');
      console.error(report);
      return;
    }
    
    console.log('\n✅ 迁移完成！所有数据已迁移到 SQLite');
    console.log('='.repeat(60));
  }
  
  private async phase0(): Promise<void> {
    // 1. 备份
    backupLocalStorage();
    console.log('  ✅ 备份完成');
    
    // 2. 等待 2 秒让用户确认
    await this.delay(2000);
  }
  
  private async phase1(): Promise<void> {
    const batches = [
      { name: '品种库', key: 'crop_varieties', storageKey: 'crop_varieties' },
      { name: '种源', key: 'seed_sources', storageKey: 'crop_seed_sources' },
      { name: '育苗', key: 'seedlings', storageKey: 'crop_seedlings' },
      { name: '种植', key: 'plantings', storageKey: 'crop_plantings' },
      { name: '采收', key: 'harvest_records', storageKey: 'harvest_records' },
      { name: '订单', key: 'crop_orders', storageKey: 'crop_orders' },
      { name: '实例', key: 'crop_instances', storageKey: 'crop_instances' },
    ];
    
    for (const batch of batches) {
      console.log(`\n  📦 迁移 ${batch.name}...`);
      const result = await this.migrateBatch(batch.storageKey, batch.key);
      if (result.success) {
        console.log(`     ✅ ${batch.name} 迁移成功`);
      } else {
        console.error(`     ❌ ${batch.name} 迁移失败: ${result.message}`);
        throw new Error(`${batch.name} 迁移失败`);
      }
    }
  }
  
  private async migrateBatch(storageKey: string, apiEndpoint: string): Promise<MigrationResult> {
    const localData = localStorage.getItem(storageKey);
    if (!localData) {
      return { success: true, message: '无数据' };
    }
    
    const items = JSON.parse(localData);
    let successCount = 0;
    let failCount = 0;
    
    for (const item of items) {
      try {
        const response = await fetch(`/api/${apiEndpoint}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(item)
        });
        
        if (response.ok) successCount++;
        else failCount++;
      } catch (e) {
        failCount++;
      }
    }
    
    return {
      success: failCount === 0,
      message: `成功 ${successCount}, 失败 ${failCount}`
    };
  }
  
  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

### 6.2 使用方法

```javascript
// 在浏览器控制台执行

executor.execute();
```
