### 2.2 核心表结构设计

#### 2.2.1 employees（员工基础信息表）

```sql
CREATE TABLE employees (
    id                    VARCHAR(36) PRIMARY KEY,          -- 员工ID (UUID)
    employee_code         VARCHAR(20) UNIQUE NOT NULL,      -- 员工编号 (EMP20260423001)
    name                  VARCHAR(50) NOT NULL,              -- 姓名
    id_card               VARCHAR(18) UNIQUE,                -- 身份证号
    phone                 VARCHAR(20) NOT NULL,              -- 手机号
    gender                ENUM('male', 'female') NOT NULL,   -- 性别
    birth_date            DATE,                              -- 出生日期
    address               VARCHAR(255),                      -- 家庭住址
    bank_account          VARCHAR(30),                       -- 银行卡号
    bank_name             VARCHAR(100),                      -- 开户行名称
    emergency_contact     VARCHAR(50),                       -- 紧急联系人
    emergency_phone       VARCHAR(20),                       -- 紧急联系人电话
    department            VARCHAR(50) NOT NULL,              -- 部门
    position              VARCHAR(50) NOT NULL,              -- 岗位
    team_id               VARCHAR(36),                       -- 班组ID (FK → teams.id)
    employee_type         ENUM('regular', 'temp') DEFAULT 'regular', -- 员工类型
    status                ENUM('probation', 'regular', 'resigned') DEFAULT 'probation', -- 状态
    entry_date            DATE NOT NULL,                     -- 入职日期
    probation_end_date    DATE,                              -- 试用期结束日期
    resign_date           DATE,                              -- 离职日期
    resign_reason         TEXT,                              -- 离职原因
    avatar_url            VARCHAR(255),                      -- 头像URL
    id_card_front_url     VARCHAR(255),                      -- 身份证正面照片
    id_card_back_url      VARCHAR(255),                      -- 身份证反面照片
    bank_card_url         VARCHAR(255),                      -- 银行卡照片
    health_cert_url       VARCHAR(255),                      -- 体检报告照片
    created_by            VARCHAR(36),                       -- 创建人ID
    updated_by            VARCHAR(36),                       -- 更新人ID
    created_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 创建时间
    updated_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, -- 更新时间
    
    INDEX idx_department (department),
    INDEX idx_team (team_id),
    INDEX idx_status (status),
    INDEX idx_employee_type (employee_type),
    INDEX idx_entry_date (entry_date),
    FOREIGN KEY (team_id) REFERENCES teams(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='员工基础信息表';
```

#### 2.2.2 attendance_records（考勤记录表）

```sql
CREATE TABLE attendance_records (
    id                    VARCHAR(36) PRIMARY KEY,          -- 记录ID (UUID)
    employee_id           VARCHAR(36) NOT NULL,              -- 员工ID (FK → employees.id)
    date                  DATE NOT NULL,                     -- 考勤日期
    schedule_id           VARCHAR(36),                       -- 排班ID (FK → schedules.id)
    status                ENUM('normal', 'late', 'early_leave', 'absent', 'rest', 'leave', 'sick_leave', 'personal_leave', 'annual_leave', 'maternity_leave', 'funeral_leave', 'marriage_leave') NOT NULL, -- 考勤状态
    check_in_time         TIME,                              -- 签到时间
    check_out_time        TIME,                              -- 签退时间
    late_minutes          INT DEFAULT 0,                     -- 迟到分钟数
    early_leave_minutes   INT DEFAULT 0,                     -- 早退分钟数
    work_hours            DECIMAL(5,2),                      -- 工作时长（小时）
    overtime_hours        DECIMAL(5,2) DEFAULT 0,            -- 加班时长（小时）
    notes                 TEXT,                              -- 备注
    approval_id           VARCHAR(36),                       -- 关联审批单ID (FK → approvals.id)
    is_manually_corrected BOOLEAN DEFAULT FALSE,             -- 是否人工修正
    corrected_by          VARCHAR(36),                       -- 修正人ID
    corrected_at          TIMESTAMP,                         -- 修正时间
    created_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 创建时间
    updated_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, -- 更新时间
    
    INDEX idx_employee_date (employee_id, date),
    INDEX idx_date (date),
    INDEX idx_status (status),
    INDEX idx_schedule (schedule_id),
    INDEX idx_approval (approval_id),
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    FOREIGN KEY (schedule_id) REFERENCES schedules(id),
    FOREIGN KEY (approval_id) REFERENCES approvals(id),
    UNIQUE KEY uk_employee_date (employee_id, date)          -- 每人每天一条记录
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='考勤记录表';
```

#### 2.2.3 salary_records（工资记录表）

```sql
CREATE TABLE salary_records (
    id                    VARCHAR(36) PRIMARY KEY,          -- 记录ID (UUID)
    employee_id           VARCHAR(36) NOT NULL,              -- 员工ID (FK → employees.id)
    period                VARCHAR(7) NOT NULL,               -- 周期 (YYYY-MM)
    status                ENUM('draft', 'pending_approval', 'approved', 'paid', 'cancelled') DEFAULT 'draft', -- 状态
    
    -- 应发项目
    base_salary           DECIMAL(10,2) DEFAULT 0,           -- 基本工资
    overtime_pay          DECIMAL(10,2) DEFAULT 0,           -- 加班工资
    piecework_pay         DECIMAL(10,2) DEFAULT 0,           -- 计件工资
    attendance_bonus      DECIMAL(10,2) DEFAULT 0,           -- 满勤奖
    position_allowance    DECIMAL(10,2) DEFAULT 0,           -- 岗位津贴
    meal_allowance        DECIMAL(10,2) DEFAULT 0,           -- 餐补
    housing_allowance     DECIMAL(10,2) DEFAULT 0,           -- 房补
    other_allowance       DECIMAL(10,2) DEFAULT 0,           -- 其他补贴
    gross_pay             DECIMAL(10,2) DEFAULT 0,           -- 应发合计
    
    -- 扣款项目
    leave_deduction       DECIMAL(10,2) DEFAULT 0,           -- 请假扣款
    late_deduction        DECIMAL(10,2) DEFAULT 0,           -- 迟到扣款
    absent_deduction      DECIMAL(10,2) DEFAULT 0,           -- 缺勤扣款
    social_insurance      DECIMAL(10,2) DEFAULT 0,           -- 社保扣款
    housing_fund          DECIMAL(10,2) DEFAULT 0,           -- 公积金
    income_tax            DECIMAL(10,2) DEFAULT 0,           -- 个人所得税
    other_deduction       DECIMAL(10,2) DEFAULT 0,           -- 其他扣款
    total_deduction       DECIMAL(10,2) DEFAULT 0,           -- 扣款合计
    
    -- 实发
    net_pay               DECIMAL(10,2) DEFAULT 0,           -- 实发工资
    
    -- 明细数据
    attendance_days       INT DEFAULT 0,                     -- 出勤天数
    leave_days            INT DEFAULT 0,                     -- 请假天数
    absent_days           INT DEFAULT 0,                     -- 缺勤天数
    overtime_hours        DECIMAL(5,2) DEFAULT 0,            -- 加班时长
    piecework_count       DECIMAL(10,2) DEFAULT 0,           -- 计件数量
    
    -- 计算标准
    daily_salary          DECIMAL(10,2),                     -- 日薪标准
    hourly_salary         DECIMAL(10,2),                     -- 时薪标准
    attendance_bonus_rule DECIMAL(10,2),                     -- 满勤奖标准
    late_deduction_rule   DECIMAL(10,2),                     -- 迟到扣款标准
    
    -- 审批信息
    approval_id           VARCHAR(36),                       -- 工资审批单ID
    approved_by           VARCHAR(36),                       -- 审批人
    approved_at           TIMESTAMP,                         -- 审批时间
    
    -- 发放信息
    pay_date              DATE,                              -- 实际发放日期
    pay_method            ENUM('bank', 'cash', 'wechat', 'alipay'), -- 发放方式
    pay_status            ENUM('unpaid', 'paid') DEFAULT 'unpaid', -- 发放状态
    
    -- 审计
    calculated_by         VARCHAR(36),                       -- 计算人
    calculated_at         TIMESTAMP,                         -- 计算时间
    created_by            VARCHAR(36),                       -- 创建人
    updated_by            VARCHAR(36),                       -- 更新人
    created_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_employee_period (employee_id, period),
    INDEX idx_period (period),
    INDEX idx_status (status),
    INDEX idx_pay_status (pay_status),
    INDEX idx_approval (approval_id),
    FOREIGN KEY (employee_id) REFERENCES employees(id),
    FOREIGN KEY (approval_id) REFERENCES approvals(id),
    UNIQUE KEY uk_employee_period (employee_id, period)      -- 每人每月一条
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='工资记录表';
```

### 5.2 工资计算数据流

```
[工资计算触发（月末）]
  │
  ↓
[获取计算参数]
  ├── 计算周期：YYYY-MM
  └── 应出勤天数：从日历表获取
  │
  ↓
[获取员工列表]
  └── 查询 employees 表 (status != 'resigned')
  │
  ↓
[对每个员工计算工资]
  │
  ├── [1] 获取薪酬标准
  │   └── 查询 contracts 表 → monthly_salary
  │
  ├── [2] 获取考勤数据
  │   └── 查询 attendance_records 表
  │       ├── COUNT(status='normal') → 出勤天数
  │       ├── COUNT(status='late') → 迟到次数
  │       ├── COUNT(status='absent') → 缺勤天数
  │       └── SUM(late_minutes) → 迟到总分钟
  │
  ├── [3] 获取加班数据
  │   └── 查询 overtime_records 表
  │       └── SUM(total_hours) → 加班总时长
  │
  ├── [4] 获取请假数据
  │   └── 查询 leave_records 表
  │       └── SUM(total_days) → 请假总天数
  │
  ├── [5] 获取计件数据
  │   └── 查询 piecework_records 表
  │       └── SUM(piecework_amount) → 计件总额
  │
  ├── [6] 计算应发项目
  │   ├── base_salary = monthly_salary / 应出勤天数 × 出勤天数
  │   ├── overtime_pay = 加班时长 × 时薪 × 加班系数
  │   ├── piecework_pay = 计件总额
  │   ├── attendance_bonus = 无迟到早退缺勤 ? 满勤奖标准 : 0
  │   └── gross_pay = base + overtime + piecework + bonus + allowances
  │
  ├── [7] 计算扣款项目
  │   ├── leave_deduction = 请假天数 × 日薪 × 扣款系数
  │   ├── late_deduction = 迟到次数 × 扣款标准
  │   ├── absent_deduction = 缺勤天数 × 日薪
  │   ├── social_insurance = 工资基数 × 社保比例
  │   └── total_deduction = SUM(各项扣款)
  │
  ├── [8] 计算实发工资
  │   └── net_pay = gross_pay - total_deduction
  │
  └── [9] 保存工资记录
      └── INSERT INTO salary_records 表
  │
  ↓
[生成工资汇总表]
  └── 按部门/班组汇总
  │
  ↓
[提交审批]
  └── 创建工资审批单 → approvals 表
  │
  ↓
[审批通过]
  │
  └── 更新工资状态 → salary_records 表 (status='approved')
  │
  ↓
[工资发放]
  │
  └── 更新发放状态 → salary_records 表 (pay_status='paid')
```
